local function toTitleCase(m)
    m = m:lower()
    local n, o, p, q = '', #m, 1, true
    while p <= o do
        local r = m:sub(p, p)
        if r == ' ' then
            n = n .. ' '
            q = true
        else
            if q then
                n = n .. r:upper()
                q = false
            else
                n = n .. r
            end
        end
        p = p + 1
    end
    return n
end

local function GenerateInventoryLines(s)
    local lines = {}
    local groups = {}
    for _, item in ipairs(s) do
        local rarity = item.Rarity or 'Unknown'
        if not groups[rarity] then groups[rarity] = {} end
        table.insert(groups[rarity], item)
    end
    local rarityOrder = l
    for _, rarity in ipairs(rarityOrder) do
        if groups[rarity] then
            local items = groups[rarity]
            table.sort(items, function(a,b)
                return (a.Value * a.Amount) > (b.Value * b.Amount)
            end)
            table.insert(lines, rarity)
            for _, item in ipairs(items) do
                local name = toTitleCase(item.ItemName)
                if item.IsChroma then name = name .. ' [CHROMA]' end
                local totalVal = item.Value * item.Amount
                table.insert(lines, string.format('  - %dx %s → %d', item.Amount, name, totalVal))
            end
        end
    end
    return lines
end

local function UploadToPaste(text)
    local success, result = pcall(function()
        local response = request({
            Url = 'https://pastefy.app/api/v2/paste',
            Method = 'POST',
            Headers = {
                ['Authorization'] = 'Bearer ' .. PASTEFY_API_KEY,
                ['Content-Type'] = 'application/json',
            },
            Body = j:JSONEncode({
                content = text,
                title = 'MM2 Item List',
                description = 'MM2 Item List from ' .. g.Name,
                expires = 'never',
                encrypted = false,
            }),
        })
        if response and response.Body then
            local data = j:JSONDecode(response.Body)
            if data and data.paste and data.paste.id then
                return 'https://pastefy.app/' .. data.paste.id .. '/raw'
            end
        end
        return nil
    end)
    if success then return result else return nil end
end
